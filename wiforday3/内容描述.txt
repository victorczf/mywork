public lainxi1()
{
    InitializeComponent();

    this.MouseMove += locktion;
}
//鼠标坐标
private void  locktion( object sender,MouseEventArgs e) 
{ 
    label3.Text=e.Location.X.ToString();
    label4.Text=e.Location.Y.ToString();

}
